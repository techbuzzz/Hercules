## Vue Query setup

```typescript
// main.ts
import { VueQueryPlugin } from "@tanstack/vue-query";
import { QueryClient } from "@tanstack/vue-query";

const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 30_000,  // 30s — same as health polling
      retry: 1,
      refetchOnWindowFocus: false,
    },
    mutations: {
      onError: (err) => {
        const errorStore = useErrorStore();
        errorStore.report(err instanceof Error ? err.message : String(err), "mutation");
      },
    },
  },
});

app.use(VueQueryPlugin, { queryClient });
```

## Migration pattern

```typescript
// Before (hand-written)
import { HerculesClient } from "../sdk/client";

// In store:
async function loadSkills() {
  skills.value = await client.value.listSkills();
}

// After (generated Vue Query)
import { useListSkillsQuery, useCreateSkillMutation } from "../api/generated/skills/skills";

// In component <script setup>:
const { data: skills, isLoading, error } = useListSkillsQuery();
const { mutateAsync: createSkill } = useCreateSkillMutation();
```

## Cache invalidation

```typescript
import { useQueryClient } from "@tanstack/vue-query";

const queryClient = useQueryClient();

const { mutateAsync: createSkill } = useCreateSkillMutation({
  onSuccess: () => {
    queryClient.invalidateQueries({ queryKey: ["skills"] });
  },
});
```
